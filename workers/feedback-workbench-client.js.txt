    function escapeHtml(value) {
        return String(value == null ? '' : value)
            .replace(/&/g, '&amp;')
            .replace(/</g, '&lt;')
            .replace(/>/g, '&gt;')
            .replace(/"/g, '&quot;')
            .replace(/'/g, '&#39;');
    }

    function paragraphs(value) {
        var blocks = String(value == null ? '' : value).split(/\n{2,}/);
        return blocks
            .filter(function (block) {
                return block.trim();
            })
            .map(function (block) {
                return '<p>' + escapeHtml(block.trim()).replace(/\n/g, '<br />') + '</p>';
            })
            .join('');
    }

    /*
     * Markdown rendering for authored bodies (§19.2).
     *
     * Agent results are written in GFM — headings, tables, fenced code — and

        br: [],
        hr: [],
        strong: [],
        b: [],
        em: [],
        i: [],
        del: [],
        s: [],
        sup: [],

        object: true,

    var MARKDOWN_URL_ATTRIBUTES = { href: true, src: true };

    function markdownUrlIsSafe(value) {
        return /^(?:https?:|mailto:|tel:|\/|#|\?)/.test(
            String(value == null ? '' : value)
                .trim()
                .toLowerCase()
        );
    }

    function unwrapElement(element) {
        var parent = element.parentNode;
        if (!parent) return;
        while (element.firstChild) parent.insertBefore(element.firstChild, element);
        parent.removeChild(element);
    }

    function sanitizeMarkdownElement(element) {
        var tag = element.tagName.toLowerCase();
        var allowedAttributes = MARKDOWN_ALLOWED_TAGS[tag] || [];

        Array.prototype.slice.call(element.attributes).forEach(function (attribute) {
            var name = attribute.name.toLowerCase();
            if (allowedAttributes.indexOf(name) < 0) {
                element.removeAttribute(attribute.name);
                return;
            }
            if (MARKDOWN_URL_ATTRIBUTES[name] && !markdownUrlIsSafe(attribute.value)) {
                element.removeAttribute(attribute.name);
            }
        });

        if (tag === 'a' && element.getAttribute('href')) {
            element.setAttribute('target', '_blank');
            element.setAttribute('rel', 'noopener noreferrer nofollow ugc');
        }

        // A checkbox that stays clickable would look like a control the reader
        // can toggle, but nothing persists it — task lists are display only.
        if (tag === 'input') {
            element.setAttribute('type', 'checkbox');
            element.setAttribute('disabled', 'disabled');
        }
    }

    function sanitizeMarkdownHtml(html) {
        var template = document.createElement('template');
        template.innerHTML = String(html == null ? '' : html);

        var walker = document.createTreeWalker(template.content, NodeFilter.SHOW_ELEMENT, null);
        var dropped = [];
        var unwrapped = [];

        while (walker.nextNode()) {
            var element = walker.currentNode;
            var tag = element.tagName.toLowerCase();

            if (MARKDOWN_DROPPED_TAGS[tag]) {
                dropped.push(element);
                continue;
            }

            sanitizeMarkdownElement(element);

            if (!MARKDOWN_ALLOWED_TAGS[tag]) unwrapped.push(element);
        }

        dropped.forEach(function (element) {
            if (element.parentNode) element.parentNode.removeChild(element);
        });
        unwrapped.forEach(unwrapElement);

        // GFM task lists arrive as a bare checkbox inside a bulleted <li>;
        // marking them lets the stylesheet drop the redundant bullet.
        Array.prototype.forEach.call(
            template.content.querySelectorAll('li > input[type="checkbox"]'),
            function (checkbox) {
                var item = checkbox.parentNode;
                item.classList.add('markdown-task-item');
                if (item.parentNode && item.parentNode.classList) {
                    item.parentNode.classList.add('markdown-task-list');
                }
            }
        );

        // A wide table cannot shrink below its content, so it needs a scroll
        // container of its own or it widens the whole card at 375px (§19.6).
        Array.prototype.forEach.call(
            template.content.querySelectorAll('table'),
            function (table) {
                if (!table.parentNode) return;
                var wrapper = document.createElement('div');
                wrapper.className = 'markdown-table';
                table.parentNode.insertBefore(wrapper, table);
                wrapper.appendChild(table);
            }
        );

        return template.innerHTML;
    }

    /* SCN-FWB-001：终端转义序列不是内容，在网页上既没有颜色也读不懂，只会把真正
       的内容埋掉（2026-09-01 用户截图：一条 integration_verification_failed 回复

       这里不可避免是第二份实现——客户端脚本是 Worker 内联的 `<script>` 字符串、
       没有模块图，import 不了执行器的 stripAnsi（与 vendored marked 同一约束）。
       正则与 packages/feedback-platform/executor/verification.js 逐字一致，
       改一处必须改另一处。 */
    var ANSI_PATTERN =
        /\u001B\[[0-9;?]*[ -/]*[@-~]|\u001B\][\s\S]*?(?:\u0007|\u001B\\)|\u001B[@-Z\\-_]/g;

    function stripAnsi(value) {
        return String(value == null ? '' : value).replace(ANSI_PATTERN, '');
    }

    function renderMarkdown(value) {
        var text = stripAnsi(value);
        if (!text.trim()) return '';

        var parser = window.marked;
        if (!parser || typeof parser.parse !== 'function') return paragraphs(text);

        try {
            return sanitizeMarkdownHtml(parser.parse(text, MARKDOWN_OPTIONS));
        } catch (error) {
            // Never let a malformed body blank out the timeline: fall back to
            // the escaped-paragraph rendering this replaced.
            return paragraphs(text);
        }
    }

    function relativeTime(value) {
        var timestamp = Date.parse(value);
        if (!timestamp) return '';
        var seconds = Math.floor((Date.now() - timestamp) / 1000);
        if (seconds < 60) return '刚刚';
        if (seconds < 3600) return Math.floor(seconds / 60) + ' 分钟前';
        if (seconds < 86400) return Math.floor(seconds / 3600) + ' 小时前';
        if (seconds < 2592000) return Math.floor(seconds / 86400) + ' 天前';
        return new Date(timestamp).toLocaleDateString('zh-CN');
    }

    function absoluteTime(value) {
        var timestamp = Date.parse(value);
        return timestamp ? new Date(timestamp).toLocaleString('zh-CN') : '';
    }

    /* ------------------------------------------------------ keyed rendering */

    /**
     * SCN-FWB-025：刷新按 key 对账，不整块重建。`container.innerHTML = ...` 会把每一
     * 个条目都换成新节点——8 秒一次的自动同步因此能在 mousedown 和 click 之间把指针
     * 下的那一项换掉，那一次点击就打在了一个已经脱离文档的节点上。未变化的条目必须
     * 保持同一个节点对象，`cache` 存的是上一轮为每个 key 渲染出的 HTML，用来判断
     * 「未变化」。（滚动位置不在此列：`innerHTML =` 是一次同步替换，中间不发生布局，
     * 实测 `scrollTop` 并不会被清零。）
     */
    function reconcileChildren(container, items, cache) {
        if (container.dataset.renderMode !== 'keyed') {
            container.innerHTML = '';
            container.dataset.renderMode = 'keyed';
            cache = Object.create(null);
        }

        var existing = Object.create(null);
        Array.prototype.forEach.call(container.children, function (node) {
            var key = node.getAttribute('data-render-key');
            if (key !== null) existing[key] = node;
        });

        var stage = document.createElement('div');
        var nextCache = Object.create(null);
        var cursor = null;

        items.forEach(function (item) {
            var node = existing[item.key];
            if (node && cache[item.key] === item.html) {
                delete existing[item.key];
            } else {
                stage.innerHTML = item.html;
                var fresh = stage.firstElementChild;
                if (!fresh) return;
                fresh.setAttribute('data-render-key', item.key);
                if (node) container.replaceChild(fresh, node);
                delete existing[item.key];
                node = fresh;
            }
            var target = cursor ? cursor.nextSibling : container.firstChild;
            if (node !== target) container.insertBefore(node, target);
            cursor = node;
            nextCache[item.key] = item.html;
        });

        Object.keys(existing).forEach(function (key) {
            var node = existing[key];
            if (node.parentNode === container) container.removeChild(node);
        });

        return nextCache;
    }

    function request(path, options) {
        var settings = options || {};
        var headers = { Accept: 'application/json' };
        if (settings.body !== undefined) headers['Content-Type'] = 'application/json';
        if (state.adminToken) {
            headers.Authorization = 'Bearer ' + state.adminToken;
        } else if (state.ownerCapability) {
            headers.Authorization = 'Bearer ' + state.ownerCapability;
        }

        return fetch(API + path, {
            method: settings.method || 'GET',
            headers: headers,
            body: settings.body === undefined ? undefined : JSON.stringify(settings.body),
        }).then(function (response) {
            return response
                .json()
                .catch(function () {
                    return {};
                })
                .then(function (payload) {
                    if (response.ok) return payload;
                    var error = new Error(payload.error || response.statusText || '请求失败');
                    error.status = response.status;
                    error.payload = payload;
                    throw error;
                });
        });
    }

    function readOwnerFromLocation() {
        var hash = window.location.hash || '';
        if (hash.indexOf('issue=') === -1) return null;

        var params = new URLSearchParams(hash.replace(/^#/, ''));
        var issueId = params.get('issue') || '';
        var capability = params.get('capability') || '';
        if (!issueId || !capability) return null;

        // §21.1 keeps the capability out of query, logs and Referer; a
        // fragment reaches none of those, but it should not sit in the address
        // bar either, because that is what people copy and paste around. The
        // issue id does stay, so the tab is identifiable in history and the
        // owner has a number to quote — the credential link itself is handed
        // out by the copy control on the owner notice instead.
        window.history.replaceState(
            {},
            '',
            window.location.pathname +
                window.location.search +
                '#issue=' +
                encodeURIComponent(issueId)
        );
        return { issueId: issueId, capability: capability };
    }

    /**
     * §19/§21.1：`#issue=<id>` 在 owner 用掉 capability 之后仍留在地址栏，管理员也会
     * 拿着这个链接过来。队列现在按优先级排序，「刚建的那条恰好在最前面」不再成立——
     * 那本来也不是打开一条指定 Issue 的方式。这里只取 id，capability 仍只走 owner 分支。
     */
    function readIssueFromLocation() {
        var hash = window.location.hash || '';
        if (hash.indexOf('issue=') === -1) return '';
        return new URLSearchParams(hash.replace(/^#/, '')).get('issue') || '';
    }

    function restoreSession() {
        var fromLocation = readOwnerFromLocation();
        if (fromLocation) {
            try {
                window.sessionStorage.setItem(OWNER_KEY, JSON.stringify(fromLocation));
            } catch (error) {
                /* private mode: session-only */
            }
        }

        var owner = fromLocation;
        if (!owner) {
            try {
                owner = JSON.parse(window.sessionStorage.getItem(OWNER_KEY) || 'null');
            } catch (error) {
                owner = null;
            }
        }

        var adminToken = '';
        try {
            adminToken = window.sessionStorage.getItem(ADMIN_TOKEN_KEY) || '';
        } catch (error) {
            adminToken = '';
        }

        if (adminToken) {
            state.adminToken = adminToken;
            state.actor = 'admin';
            state.selectedKey = readIssueFromLocation();
        } else if (owner && owner.issueId && owner.capability) {
            state.ownerIssueId = owner.issueId;
            state.ownerCapability = owner.capability;
            state.selectedKey = owner.issueId;
            state.actor = 'owner';
        }
    }

            tab.setAttribute('aria-current', isActive ? 'page' : 'false');
        });

        // 两个设置页是懒加载的，首次切过去要等一次请求。不标出来的话这段等待
        // 看起来就是"tab 点了但页面是空的"。
        if (active === 'automations' && !state.automation) {
            setBusy($('settingsView'), true);
            loadAutomation();
        }
        if (active === 'runners' && !state.runners) {
            setBusy($('runnersView'), true);
            loadRunners();
        }
    }

    /* ----------------------------------------------------------- issue list */

    /* §19.1 的三个筛选（等我/处理中/全部）是同一批 Issue 的子集——服务端也是先取
       队列再按 `status` 过滤。放在客户端做，切换筛选就从一次约 1 秒的往返变成一次
       本地重绘；判据与 Worker 的 FEEDBACK_ATTENTION_STATUSES / FEEDBACK_ACTIVE_STATUSES
       同名同义，两边不一致会让同一个状态在队列里时隐时现。 */
    var ATTENTION_STATUSES = ['ready_for_deploy', 'needs_human', 'test_failed'];
    var ACTIVE_STATUSES = ['queued', 'in_progress', 'testing'];

    function matchesFilter(issue, filter) {
        if (filter === 'attention') return ATTENTION_STATUSES.indexOf(issue.status) !== -1;
        if (filter === 'active') return ACTIVE_STATUSES.indexOf(issue.status) !== -1;
        return true;
    }

    function visibleIssues() {
        var query = state.query.trim().toLowerCase();
        return state.issues.filter(function (issue) {
            if (!matchesFilter(issue, state.filter)) return false;
            if (!query) return true;
            return (
                String(issue.title || '')
                    .toLowerCase()
                    .indexOf(query) !== -1 || issueNumber(issue.key).indexOf(query) !== -1
            );
        });
    }

    function issueItemHtml(issue) {
        var tone = statusTone(issue.status);
        return (
            '<button class="issue-item' +
            (issue.key === state.selectedKey ? ' active' : '') +
            '" type="button" role="option" aria-selected="' +
            (issue.key === state.selectedKey) +
            '" data-issue="' +
            escapeHtml(issue.key) +
            '">' +
            '<div class="issue-item-title"><span>' +
            escapeHtml(issue.title || '(无标题)') +
            '</span><span class="badge ' +
            tone +
            '">' +
            escapeHtml(statusLabel(issue.status)) +
            '</span></div>' +
            '<div class="issue-item-copy">' +
            escapeHtml(issue.descriptionPreview || '') +
            '</div>' +
            '<div class="issue-item-meta"><span class="inline" style="gap:5px">' +
            ICON_ISSUE +
            ' #' +
            escapeHtml(issueNumber(issue.key)) +
            ' · ' +
            escapeHtml(BUSINESS_TYPE_LABELS[issue.businessType] || '待分类') +
            '</span><span>' +
            escapeHtml(relativeTime(issue.updatedAt || issue.receivedAt)) +
            '</span></div>' +
            '</button>'
        );
    }

    /* 队列可能长过服务端一页。总数按 `totals` 走全表口径，列表里放得下多少是另一回事——
       两者不一致时必须说出来，否则「等我 12」配着 9 条列表就像系统吞了 3 条。 */
    function filterTotal(filter) {
        if (!state.totals) return null;
        var total = state.totals[filter];
        return typeof total === 'number' ? total : null;
    }

    function renderQueueTruncation(visibleCount) {
        var notice = $('queueTruncation');
        var total = filterTotal(state.filter);
        // 只有整份队列没取全时才可能有漏网；取全了列表就是全部。
        var missing = state.listComplete || total === null ? 0 : total - visibleCount;
        if (missing <= 0 || state.query.trim()) {
            setHidden(notice, true);
            notice.textContent = '';
            return;
        }
        setHidden(notice, false);
        notice.textContent = '队列还有 ' + missing + ' 条未载入（本页最多 ' + PAGE_LIMIT + ' 条）';
    }

    function renderIssueList() {
        var list = $('issueList');
        var visible = visibleIssues();

        var attention = filterTotal('attention');
        $('queueAttentionBadge').textContent =
            '需你处理 ' + (attention === null ? state.attentionCount : attention);
        $('issueCount').textContent = String(visible.length);
        setHidden($('issueCount'), visible.length === 0);
        setBusy(list, state.listLoading && !state.issues.length);
        renderQueueTruncation(visible.length);

        if (!visible.length) {
            state.renderedIssueItems = Object.create(null);
            renderPlaceholder(
                list,
                state.listLoading && !state.issues.length
                    ? skeletonHtml(4)
                    : '<div style="padding:28px 16px;text-align:center;color:var(--muted)">没有匹配的 Issue</div>'
            );
            return;
        }

        state.renderedIssueItems = reconcileChildren(
            list,
            visible.map(function (issue) {
                return { key: issue.key, html: issueItemHtml(issue) };
            }),
            state.renderedIssueItems
        );
    }

    function loadIssues() {
        if (state.actor !== 'admin') return Promise.resolve();

        // 永远取整个队列：筛选在本地做，所以三个 chip 共用这一份数据，切换不再发请求。
        // 取满服务端允许的一页，让本地筛选尽量覆盖整个队列。
        state.listLoading = true;
        renderIssueList();
        return request('/api/feedback/issues?filter=all&limit=' + PAGE_LIMIT)
            .then(function (payload) {
                state.issues = payload.issues || [];
                state.attentionCount = Number(payload.attentionCount) || 0;
                state.totals = payload.totals || null;
                state.listComplete = payload.listComplete !== false;
                state.listLoading = false;
                renderIssueList();

                // §19.1: refreshing the list must not move the current selection.
                var stillVisible = state.issues.some(function (issue) {
                    return issue.key === state.selectedKey;
                });
                // 「第一条」始终指当前筛选下看得见的第一条：服务端过滤时 `state.issues`
                // 本来就只有可见的那些，本地过滤后不能退回整个队列去选，否则详情区会
                // 停在一条列表里根本没有的 Issue 上。
                if (!state.selectedKey || (!stillVisible && !state.issue)) {
                    var first = visibleIssues()[0];
                    if (first) return selectIssue(first.key);
                }
                return null;
            })
            .catch(function (error) {
                state.listLoading = false;
                renderIssueList();
                return handleAuthError(error);
            });
    }

    /* --------------------------------------------------------- issue detail */

    /**
     * 创建那条卡片上只挂「提交时就带着的」附件。后来在回复里补的附件已经各自挂在
     * 对应的评论事件上，两处都渲染就会重复出现，所以这里按事件上的附件 id 排掉。
     */
    function creationOnlyAttachments(issue) {
        if (!issue || !issue.attachments) return [];
        var commentAttachmentIds = {};
        (state.events || []).forEach(function (timelineEvent) {
            (timelineEvent.attachments || []).forEach(function (attachment) {
                commentAttachmentIds[attachment.id] = true;
            });
        });
        return issue.attachments.filter(function (attachment) {
            return !commentAttachmentIds[attachment.id];
        });
    }

    function renderTimelineEntry(event, issue) {
        var isEventRow =
            ['status.changed', 'issue.updated', 'issue.reopened', 'classification.changed'].indexOf(
                event.type
            ) !== -1;

        if (isEventRow) {
            var changeText = Object.keys(event.changes || {})
                .map(function (field) {
                    var pair = event.changes[field] || [];
                    if (field === 'status') {
                        return (
                            '将状态改为 <span class="badge ' +
                            statusTone(pair[1]) +
                            '">' +
                            escapeHtml(statusLabel(pair[1])) +
                            '</span>'
                        );
                    }
                    return (
                        '更新' +
                        escapeHtml(FIELD_LABELS[field] || field) +
                        '为 <strong>' +
                        escapeHtml(fieldValueLabel(field, pair[1])) +
                        '</strong>'
                    );
                })
                .join('，');
            if (!changeText) changeText = EVENT_FALLBACK_TEXT[event.type] || '更新了这条反馈';

            return (
                '<div class="timeline-entry"><div class="avatar event" aria-hidden="true"></div>' +
                '<div class="event-row"><span><strong>' +
                escapeHtml(actorName(event.actorType)) +
                '</strong> ' +
                (changeText || escapeHtml(event.type)) +
                // SCN-FWB-036: 回应等待时写的补充说明就存在这条 status.changed 上，
                // 而下一轮 Run 的 context 读的正是它。只渲染「将状态改为已排队」会让
                // 用户看不到自己刚说过的话，也看不出这一轮和上一轮到底有什么不同。
                (event.text
                    ? '<span class="event-note">' + escapeHtml(event.text) + '</span>'
                    : '') +
                '</span><span title="' +
                escapeHtml(absoluteTime(event.occurredAt)) +
                '">' +
                escapeHtml(relativeTime(event.occurredAt)) +
                '</span></div></div>'
            );
        }

        var isAgent = event.actorType === 'agent';
        var isCreation = event.type === 'issue.created';
        var needsAttention = event.type === 'waiting_human';
        var avatarClass = isAgent ? 'avatar ai' : 'avatar';
        var avatarText = isAgent ? 'AI' : event.actorType === 'admin' ? '管' : '你';
        var badge = isCreation
            ? '<span class="badge blue">作者</span>'
            : needsAttention
              ? '<span class="badge orange">需要你处理</span>'
              : isAgent
                ? '<span class="badge purple">自动处理</span>'
                : event.actorType === 'admin'
                  ? '<span class="badge">管理员</span>'
                  : '';
        var headline = isCreation
            ? '提交了反馈'
            : needsAttention
              ? '需要你补充信息'
              : event.type === 'artifact.created'
                ? '发布了验证证据'
                : isAgent
                  ? '发布了处理结果'
                  : '回复了';
        // 只有 `issue.created` 那一条要算这个。此前它写在每条事件的渲染里，而它自己
        // 又要扫一遍整条时间线——47 条事件跑 47 遍全表扫描。数量级上是 O(n²)，只是
        // 时间线还短的时候量不出来。
        var creationAttachments = isCreation ? creationOnlyAttachments(issue) : [];
        var bodyHtml = isCreation
            ? renderMarkdown(issue && issue.description) +
              attachmentsHtml({ attachments: creationAttachments })
            : renderMarkdown(event.text || event.title) +
              attachmentsHtml({ attachments: event.attachments });
        var evidenceHtml =
            resultEvidenceHtml(event.resultEvidence) + artifactEvidenceHtml(event.artifact);
        var visibleBodyHtml = bodyHtml || paragraphs(eventStatusText(event));

        return (
            '<article class="timeline-entry"><div class="' +
            avatarClass +
            '">' +
            escapeHtml(avatarText) +
            '</div>' +
            '<div class="comment-card' +
            (needsAttention ? ' attention' : '') +
            '"><div class="comment-head"><span><strong>' +
            escapeHtml(actorName(event.actorType)) +
            '</strong> ' +
            headline +
            ' · <span title="' +
            escapeHtml(absoluteTime(event.occurredAt)) +
            '">' +
            escapeHtml(relativeTime(event.occurredAt)) +
            '</span></span>' +
            badge +
            '</div><div class="comment-body">' +
            visibleBodyHtml +
            evidenceHtml +
            (event.mention
                ? '<p class="help">已唤醒 <code>' + escapeHtml(event.mention) + '</code></p>'
                : '') +
            '</div></div></article>'
        );
    }

    // SCN-FWB-030: a phase event whose text is "进入下一阶段" answers nothing —
    // the point of reporting progress is saying which step is running, because
    // browser verification alone can take 20 minutes.
    var RUN_PHASE_TEXT = {
        analyzing: '正在阅读代码并分析问题。',
        implementing: '正在修改代码。',
        testing: '正在运行单元测试与构建。',
        browser_verification: '正在运行浏览器回归验证，这一步通常最久。',
    };

    function runPhaseText(event) {
        var phase = (event && event.phase) || '';
        return RUN_PHASE_TEXT[phase] || '处理任务已进入下一阶段。';
    }

    function eventStatusText(event) {
        if (event.type === 'run.phase_changed') return runPhaseText(event);
        var labels = {
            'run.started': '处理任务已启动，正在读取上下文并执行。',
            'run.completed': '处理任务已完成，请核对下方结果与验证证据。',
            'run.failed': '处理任务未完成，请查看下方失败原因与验证结果。',
            'agent.message': '处理者未返回可显示的回复。',
            waiting_human: '处理暂停，需要你补充信息后才能继续。',
            'artifact.created': '已保存本次处理的验证证据。',
            'automation.suppressed': '本次自动处理被规则或配额阻止。',
            'security.blocked': '本次操作被安全策略阻止。',
        };
        return labels[event.type] || EVENT_FALLBACK_TEXT[event.type] || '该事件已记录。';
    }

    function verificationRowHtml(label, value) {
        var step = value || {};
        if (!step.required) {
            return (
                '<div class="result-check skipped"><span aria-hidden="true">—</span><div><strong>' +
                escapeHtml(label) +
                '</strong><code>' +
                escapeHtml(step.command || '') +
                '</code></div><span>不适用</span></div>'
            );
        }
        return (
            '<div class="result-check ' +
            (step.passed ? 'passed' : 'failed') +
            '"><span aria-hidden="true">' +
            (step.passed ? ICON_CHECK : '×') +
            '</span><div><strong>' +
            escapeHtml(label) +
            '</strong><code>' +
            escapeHtml(step.command || '') +
            '</code></div><span>' +
            (step.passed ? '已通过' : '未通过') +
            '</span></div>'
        );
    }

    /* SCN-FWB-048: 验证结论的唯一渲染实现，时间线的「处理结果」与候选实现卡共用。
       候选卡此前自己 `String(verification[name])`——对象求字符串得到
       `targetedTests [object Object]`，而审核候选正是管理员决定批不批准的那一屏。
       调用方必须自己先判断「有没有验证结果」：这里对空对象会渲染成三行「不适用」，
       那是在说这三步不需要跑，与「还没跑」是两回事。 */
    function verificationChecksHtml(value) {
        var verification = value || {};
        var visual = verification.visualEvidence || {};
        return (
            '<div class="result-checks">' +
            verificationRowHtml('目标测试', verification.targetedTests) +
            verificationRowHtml('构建', verification.build) +
            verificationRowHtml('浏览器验证', verification.playwright) +
            '</div>' +
            (visual.required
                ? '<p class="visual-status ' +
                  (visual.present ? 'passed' : 'failed') +
                  '">' +
                  (visual.present ? '已附带视觉验证证据。' : '缺少必需的视觉验证证据。') +
                  '</p>'
                : '')
        );
    }

    function resultEvidenceHtml(value) {
        if (!value || typeof value !== 'object') return '';
        var files = Array.isArray(value.changedFiles) ? value.changedFiles : [];
        var verification = value.verification || {};
        var hasRequiredVerification = Boolean(
            (verification.targetedTests || {}).required ||
            (verification.build || {}).required ||
            (verification.playwright || {}).required
        );
        var changeMeta = value.changeCommit
            ? '<span>候选提交 <code>' +
              escapeHtml(String(value.changeCommit).slice(0, 12)) +
              '</code></span>'
            : '';
        var fileHtml = files.length
            ? '<div class="result-files"><strong>变更文件（' +
              files.length +
              '）</strong><ul>' +
              files
                  .map(function (path) {
                      return '<li><code>' + escapeHtml(path) + '</code></li>';
                  })
                  .join('') +
              '</ul></div>'
            : hasRequiredVerification
              ? '<p class="result-warning">没有记录任何变更文件，不能据此判断问题已处理。</p>'
              : '<p class="help">本次为只读分析，没有修改仓库文件。</p>';

        // A gate rejection is only actionable when it names the rule and the
        // path it fired on. Without this the Issue showed "被质量门禁阻断" and
        // nothing else, so the same Run could only fail the same way again.
        var violations = Array.isArray(value.violations) ? value.violations : [];
        var violationHtml = violations.length
            ? '<div class="result-files result-violations"><strong>被质量门禁拒绝（' +
              violations.length +
              '）</strong><ul>' +
              violations
                  .map(function (violation) {
                      return (
                          '<li>' +
                          escapeHtml(String(violation.code || 'UNKNOWN_RULE')) +
                          (violation.file
                              ? ' <code>' + escapeHtml(String(violation.file)) + '</code>'
                              : '') +
                          (violation.detail
                              ? '（' + escapeHtml(String(violation.detail)) + '）'
                              : '') +
                          '</li>'
                      );
                  })
                  .join('') +
              '</ul></div>'
            : '';

        return (
            '<section class="result-evidence" data-result-evidence><div class="result-title"><strong>处理结果</strong>' +
            changeMeta +
            '</div>' +
            fileHtml +
            violationHtml +
            verificationChecksHtml(verification) +
            '</section>'
        );
    }

    function artifactEvidenceHtml(artifact) {
        if (!artifact || typeof artifact !== 'object') return '';
        var url = safeExternalUrl(artifact.url);
        var name = artifact.name || '验证证据';
        var preview =
            url && artifact.previewable
                ? '<a class="artifact-preview-link" href="' +
                  escapeHtml(url) +
                  '" target="_blank" rel="noopener noreferrer"><img data-artifact-preview src="' +
                  escapeHtml(url) +
                  '" alt="' +
                  escapeHtml(name) +
                  '" loading="lazy"></a>'
                : '';
        var link = url
            ? '<a class="artifact-link" href="' +
              escapeHtml(url) +
              '" target="_blank" rel="noopener noreferrer">打开证据</a>'
            : '<span class="help">证据链接不可用</span>';

        return (
            '<section class="artifact-evidence"><div><strong>' +
            escapeHtml(name) +
            '</strong><span>' +
            escapeHtml(artifact.type || 'artifact') +
            '</span></div>' +
            link +
            preview +
            '</section>'
        );
    }

    function fieldValueLabel(field, value) {
        var text = String(value == null ? '' : value);
        if (field === 'priority') return PRIORITY_LABELS[text] || text;
        if (field === 'ai.businessType') return BUSINESS_TYPE_LABELS[text] || text;
        if (field === 'ai.scope') return SCOPE_LABELS[text] || text;
        return text || '（空）';
    }

    function actorName(actorType) {
        if (actorType === 'agent') return 'AI 处理者';
        if (actorType === 'admin') return '管理员';
        if (actorType === 'system') return '系统';
        return '你';
    }

    function attachmentsHtml(issue) {
        var attachments = (issue && issue.attachments) || [];
        if (!attachments.length) return '';

        return attachments
            .map(function (attachment) {
                var name = escapeHtml(attachment.name || '附件');
                var meta = escapeHtml(
                    (attachment.type || '文件') + ' · ' + formatBytes(attachment.size)
                );
                var inner =
                    '<div class="attachment-thumb" aria-hidden="true">' +
                    ICON_ATTACHMENT +
                    '</div><div><strong>' +
                    name +
                    '</strong><span>' +
                    meta +
                    '</span></div>';
                return attachment.url
                    ? '<a class="attachment" href="' +
                          escapeHtml(attachment.url) +
                          '" target="_blank" rel="noopener noreferrer">' +
                          inner +
                          '</a>'
                    : '<div class="attachment">' + inner + '</div>';
            })
            .join('');
    }

    function renderTimeline() {
        var timeline = $('timeline');
        if (!state.issue) {
            state.renderedTimeline = Object.create(null);
            renderPlaceholder(
                timeline,
                state.detailLoading
                    ? skeletonHtml(5)
                    : '<div style="padding:28px 16px;text-align:center;color:var(--muted)">选择左侧队列中的 Issue 查看处理时间线</div>'
            );
            return;
        }

        var issue = state.issue;
        state.renderedTimeline = reconcileChildren(
            timeline,
            state.events.map(function (event, index) {
                return {
                    key: String(event.id || 'seq-' + (event.sequence || index)),
                    html: renderTimelineEntry(event, issue),
                };
            }),
            state.renderedTimeline
        );
    }

                : '');
    }

    /* SCN-FWB-037: 下一步选项。优先用本轮分析自己提议的（带这条 Issue 的上下文，
       "按结论删掉基线纵切面，含一条迁移测试" 远比 "补充信息并重新分析" 有用），
       模型没给或给的被状态机筛掉时，回落到服务端固定的那套。
       服务端已按 allowedReturnStates 裁过一遍，这里只负责挑一份来渲染。 */
    var NEXT_STEP_FALLBACK = {
        implement: { label: '采纳分析，开始实施', detail: '按上面的结论直接修改代码，完成后提交候选实现给你审。' },
        clarify: { label: '补充信息并重新分析', detail: '把你要的效果写清楚，带着新信息重跑一轮。' },
        close: { label: '关闭 Issue', detail: '这条不做了。' },
    };

    function nextStepOptions(action) {
        var proposed = [];
        (state.events || []).forEach(function (event) {
            if (!event.nextSteps || !event.nextSteps.length) return;
            // 只认产生当前这条等待的那一轮：上一轮的建议早就过期了。
            if (action.runId && event.runId && event.runId !== action.runId) return;
            proposed = event.nextSteps;
        });

        var isAdmin = state.actor === 'admin';
        var allowed = action.allowedReturnStates || [];
        var serverDefaults = [{ action: 'implement' }, { action: 'clarify' }, { action: 'close' }];

        function usableOptions(source) {
            return source
                .filter(function (option) {
                    var fallback = NEXT_STEP_FALLBACK[option.action];
                    if (!fallback) return false;
                    // 采纳与关闭都是管理员的决定（§21.3 / EXC-FWB-003）。
                    if (option.action !== 'clarify' && !isAdmin) return false;
                    var decision = option.action === 'close' ? 'closed' : 'queued';
                    return allowed.indexOf(decision) !== -1;
                })
                .map(function (option) {
                    var fallback = NEXT_STEP_FALLBACK[option.action];
                    return {
                        action: option.action,
                        decision: option.action === 'close' ? 'closed' : 'queued',
                        policyDecision: option.action === 'implement' ? 'implement' : '',
                        needsNote: option.action === 'clarify',
                        label: option.label || fallback.label,
                        detail: option.detail || fallback.detail,
                    };
                });
        }

        var options = usableOptions(proposed.length ? proposed : serverDefaults);
        // Agent 的提议可能整组被权限筛掉（比如 owner 打开一张只提议「采纳」的卡）。
        // 这个分支渲染的是卡片上仅有的按钮——空列表等于把这张卡变成死路，
        // 所以筛空时回落到固定选项，让「补充信息」这条路永远在。
        if (!options.length && proposed.length) options = usableOptions(serverDefaults);
        return options;
    }

    function nextStepChoicesHtml(action) {
        var options = nextStepOptions(action);
        if (!options.length) return '';

        return (
            '<div class="next-step-choices">' +
            options
                .map(function (option) {
                    return (
                        '<button class="next-step-choice' +
                        (option.action === 'implement' ? ' primary' : '') +
                        (option.action === 'close' ? ' danger' : '') +
                        '" type="button" data-human-action="' +
                        escapeHtml(action.id) +
                        '" data-decision="' +
                        escapeHtml(option.decision) +
                        '"' +
                        (option.policyDecision
                            ? ' data-policy-decision="' + escapeHtml(option.policyDecision) + '"'
                            : '') +
                        (option.needsNote ? ' data-needs-note="1"' : '') +
                        '><span class="next-step-label">' +
                        escapeHtml(option.label) +
                        '</span>' +
                        (option.detail
                            ? '<span class="next-step-detail">' +
                              escapeHtml(option.detail) +
                              '</span>'
                            : '') +
                        '</button>'
                    );
                })
                .join('') +
            '</div>'
        );
    }

    /* §16.3: an evidence item is `{label, summary}`. Historical rows carry raw
       diagnostic objects instead (`{policy, runId}`, `{executorId, epoch}`…),
       and rendering only `label`/`summary` turned every one of them into an
       empty box under a correct "N 项" count — worse than showing nothing,
       because it reads as "you missed something". Those rows are already in
       production D1 and are never rewritten, so the client degrades to showing
       the raw key/value rather than a blank. */
    function evidenceLabel(item) {
        if (!item || typeof item !== 'object') return '证据';
        if (item.label || item.type) return item.label || item.type;
        var keys = Object.keys(item);
        return keys.length ? keys[0] : '证据';
    }

    function evidenceSummary(item) {
        if (item === null || item === undefined) return '';
        if (typeof item !== 'object') return String(item);
        if (item.summary || item.value) return item.summary || item.value;
        return Object.keys(item)
            .map(function (key) {
                var value = item[key];
                return (
                    key +
                    '=' +
                    (value && typeof value === 'object' ? JSON.stringify(value) : String(value))
                );
            })
            .join('；');
    }

    /* SCN-FWB-038：`test_failed` 盖着两种处境，对它们说同一句话必然有一种是谎报。

    function ownerReturnLink() {
        return (
            window.location.origin +
            window.location.pathname +

    function copyOwnerReturnLink() {
        var input = $('ownerLinkInput');
        if (!input) return;

        // Select first: that leaves the manual Ctrl/⌘ + C path ready whenever
        // the async clipboard write is unavailable, denied or rejected.
        input.focus();
        input.select();
        input.setSelectionRange(0, input.value.length);

        var clipboard = window.navigator.clipboard;
        if (!clipboard || typeof clipboard.writeText !== 'function') {
            toastTimer('已选中链接，请按 Ctrl/⌘ + C 复制');
            return;
        }

        clipboard
            .writeText(input.value)
            .then(function () {
                toastTimer('回访链接已复制，请妥善保存');
            })
            .catch(function () {
                toastTimer('已选中链接，请按 Ctrl/⌘ + C 复制');
            });
    }

    function renderComposer() {
        var issue = state.issue;
        setHidden($('composer'), !issue || state.actor === 'anonymous');
        if (!issue) return;

        var isAdmin = state.actor === 'admin';
        var isTerminal = TERMINAL_STATUSES.indexOf(issue.status) !== -1;
        var canResume = isAdmin ? !isTerminal : issue.status === 'needs_human' && state.humanAction;

        var options = [];
        if (!isAdmin && canResume) {
            options.push({
                value: 'resume',
                label: '回复并交给 ' + defaultProviderLabel() + ' 继续处理',
            });
        }
        options.push({ value: 'record', label: '仅记录回复，暂不触发' });
        if (isAdmin && canResume) {
            options.push({
                value: 'resume',
                label: '回复并交给 ' + defaultProviderLabel() + ' 继续处理',
            });
        }
        if (isAdmin && !isTerminal) options.push({ value: 'close', label: '回复并关闭 Issue' });

        var select = $('replyMode');
        var selectedMode = state.replyMode || select.value;
        select.innerHTML = options
            .map(function (option) {
                return (
                    '<option value="' + option.value + '">' + escapeHtml(option.label) + '</option>'
                );
            })
            .join('');
        if (
            options.some(function (option) {
                return option.value === selectedMode;
            })
        ) {
            select.value = selectedMode;
        }
        state.replyMode = select.value;

        $('mentionBar').innerHTML = isAdmin
            ? '<button class="mention-button" type="button" data-mention="@codex-agent">@ Codex Agent</button>' +
              '<button class="mention-button" type="button" data-mention="@claude-agent">@ Claude Agent</button>' +
              '<span class="mention-hint">提及处理者会立即触发新的处理任务</span>'
            : canResume
              ? '<span class="mention-hint">回复会恢复当前等待中的处理任务</span>'
              : '<span class="mention-hint">你的回复会记录在时间线上</span>';

        renderReplyAttachments();
        updateRouteHint();
    }

    function renderReplyAttachments() {
        var list = $('replyAttachmentList');
        var summary = $('replyAttachmentSummary');
        if (!list || !summary) return;
        if (!state.replyAttachments.length) {
            list.innerHTML = '';
            setHidden(list, true);
            summary.textContent = '最多 5 个，单个不超过 4 MiB';
            return;
        }
        setHidden(list, false);
        list.innerHTML = state.replyAttachments
            .map(function (attachment) {
                return (
                    '<span class="reply-attachment-chip"><strong>' +
                    escapeHtml(attachment.name) +
                    '</strong><span>' +
                    formatBytes(attachment.size) +
                    '</span></span>'
                );
            })
            .join('');
        if (
            options.some(function (option) {
                return option.value === selectedMode;
            })
        ) {
            select.value = selectedMode;
        }
        summary.textContent = state.replyAttachments.length + ' 个附件已选';
    }

    function readReplyAttachment(file) {
        return new Promise(function (resolve, reject) {
            var reader = new FileReader();
            reader.onload = function () {
                resolve({
                    name: file.name,
                    type: file.type || 'application/octet-stream',
                    size: file.size,
                    dataUrl: reader.result,
                });
            };
            reader.onerror = reject;
            reader.readAsDataURL(file);
        });
    }

    function estimatedReplyAttachmentBytes(files) {
        return files.reduce(function (total, file) {
            var dataUrlPrefix = 'data:' + (file.type || 'application/octet-stream') + ';base64,';
            return total + new Blob([file.name, dataUrlPrefix]).size + Math.ceil(file.size / 3) * 4;
        }, 0);
    }

    function defaultProviderLabel() {
        var provider = (state.runners && state.runners.defaultProvider) || 'codex';
        return PROVIDER_LABELS[provider] || provider;
    }

    /* §19.3: always show which provider and policy the reply will wake. */
    function updateRouteHint() {
        var mode = $('replyMode').value;
        var text = String($('replyInput').value || '').toLowerCase();
        var provider = '';
        Object.keys(PROVIDER_MENTIONS).forEach(function (id) {
            if (text.indexOf(PROVIDER_MENTIONS[id]) !== -1) provider = id;
        });

        if (mode !== 'resume') {
            $('replyRouteHint').textContent =
                mode === 'close'
                    ? '提交后写入回复并关闭 Issue，不唤醒处理者。'
                    : '仅写入时间线，不唤醒处理者。';
            return;
        }

        var label = provider ? PROVIDER_LABELS[provider] : defaultProviderLabel();
        var source = provider ? '（由 @mention 指定）' : '（项目默认执行器）';
        $('replyRouteHint').textContent =
            '提交后将唤醒 ' + label + ' ' + source + '，策略由 Issue 分类决定，不由回复内容指定。';
    }

    function applyIssueSnapshot(payload, generation) {
        if (!payload || (generation && generation !== state.selectionGeneration)) return;
        var issue = payload.issue || {};
        // Admin detail nests workflow fields; flatten to one shape.
        state.issue = issue.workflow
            ? {
                  key: issue.key,
                  title: issue.title,
                  description: issue.description,
                  receivedAt: issue.receivedAt,
                  version: issue.version || payload.version,
                  attachments: issue.attachments || [],
                  businessType: (issue.ai || {}).businessType,
                  scope: (issue.ai || {}).scope,
                  status: issue.workflow.status,
                  priority: issue.workflow.priority,
                  assignee: issue.workflow.assignee,
                  updatedAt: issue.workflow.updatedAt,
                  pagePath: safePagePath((issue.context || {}).url),
              }
            : issue;
        if (!state.issue.version) state.issue.version = payload.version;
        state.events = payload.events || [];
        state.humanAction =
            (payload.humanActions || []).filter(function (action) {
                return action.status === 'active';
            })[0] || null;
        state.designs = payload.designs || [];
        state.candidates = payload.candidates || [];
        state.releases = payload.releases || [];
        if (payload.attachmentAccessExpiresAt) {
            state.attachmentAccessExpiresAt = Date.parse(payload.attachmentAccessExpiresAt) || 0;
        }
        state.detailLoading = false;
        setBusy($('timeline'), false);
        renderIssueDetail();
    }

    /**
     * 打开另一条 Issue 要等一次聚合请求。在它回来之前把详情区留在上一条 Issue 上，
     * 是最容易被读成"点了没反应"的一种停顿——用户看到的标题、时间线、下一步全都
     * 还是旧的。这里先立刻切到目标标题加骨架，等快照到了再填真内容。
     */
    function showDetailLoading(key) {
        var summary = state.issues.filter(function (issue) {
            return issue.key === key;
        })[0];
        state.detailLoading = true;
        state.issue = null;
        state.events = [];
        state.humanAction = null;
        state.designs = [];
        state.candidates = [];
        state.releases = [];
        renderIssueDetail();
        if (summary) {
            $('issueTitle').textContent = summary.title || '(无标题)';
        }
        setBusy($('timeline'), true);
    }

    function selectIssue(key) {
        var switching = key !== state.selectedKey;
        if (switching) {
            state.replyRequestId = '';
            state.replyMode = '';
            state.attachmentAccessExpiresAt = 0;
            state.syncUnsupported = false;
        }
        state.selectedKey = key;
        var generation = ++state.selectionGeneration;
        renderIssueList();
        if (switching) showDetailLoading(key);
        var base = '/api/feedback/issues/' + encodeURIComponent(key);
        return request(base + '/snapshot')
            .then(function (payload) {
                applyIssueSnapshot(payload, generation);
                startAutoRefresh();
                return payload;
            })
            .catch(function (error) {
                // Older Workers can still serve the six sub-resource routes;
                // retain a compatibility fallback while the new endpoint rolls out.
                if (error && error.status === 404) {
                    return Promise.all([
                        request(base),
                        request(base + '/events'),
                        request(base + '/human-actions'),
                        request(base + '/designs'),
                        request(base + '/candidates'),
                        request(base + '/releases'),
                    ]).then(function (results) {
                        applyIssueSnapshot(
                            {
                                issue: results[0].issue,
                                events: results[1].events,
                                version: results[1].version,
                                humanActions: results[2].humanActions,
                                designs: results[3].designs,
                                candidates: results[4].candidates,
                                releases: results[5].releases,
                            },
                            generation
                        );
                        startAutoRefresh();
                        return null;
                    });
                }
                state.detailLoading = false;
                setBusy($('timeline'), false);
                renderIssueDetail();
                handleAuthError(error);
                return null;
            });
    }

    function pollCurrentIssue() {
        if (
            state.pollInFlight ||
            !state.issue ||
            !state.selectedKey ||
            state.syncUnsupported ||
            document.hidden ||
            !navigator.onLine ||
            state.submitting
        ) {
            return Promise.resolve(null);
        }
        state.pollInFlight = true;
        var generation = state.selectionGeneration;
        var base = '/api/feedback/issues/' + encodeURIComponent(state.selectedKey);
        var version = state.issue.version || 0;
        var attachmentAccessExpired =
            state.attachmentAccessExpiresAt > 0 &&
            state.attachmentAccessExpiresAt <= Date.now() + 30 * 1000;
        var syncPath = attachmentAccessExpired
            ? base + '/sync'
            : base + '/sync?version=' + encodeURIComponent(version);
        return request(syncPath)
            .then(function (payload) {
                if (generation !== state.selectionGeneration) return payload;
                if (payload && payload.changed) {
                    applyIssueSnapshot(payload, generation);
                    if (state.actor === 'admin') return loadIssues();
                }
                return payload;
            })
            .catch(function (error) {
                if (error && error.status === 401) stopAutoRefresh();
                if (error && error.status === 404) {
                    state.syncUnsupported = true;
                    stopAutoRefresh();
                }
                return null;
            })
            .finally(function () {
                state.pollInFlight = false;
            });
    }

    function stopAutoRefresh() {
        if (state.pollHandle) window.clearInterval(state.pollHandle);
        state.pollHandle = 0;
    }

    function startAutoRefresh() {
        stopAutoRefresh();
        if (state.syncUnsupported || !navigator.onLine) return;
        state.pollHandle = window.setInterval(pollCurrentIssue, 8000);
    }

    function deliveryStatusText(delivery) {
        var attempts = Math.max(delivery.attemptCount, 1);
        if (delivery.status === 'succeeded') {
            return delivery.responseStatus + ' · 第 ' + attempts + ' 次尝试';
        }
        if (delivery.status === 'pending') {
            return delivery.attemptCount
                ? '重试中 · 已尝试 ' + attempts + ' 次 · ' + (delivery.lastError || '')
                : '待投递 · 已入队';
        }
        if (delivery.status === 'dead_letter') {
            return '进入 DLQ · ' + (delivery.lastError || '重试耗尽');
        }
        if (delivery.status === 'failed') {
            return '不可重试 · ' + (delivery.lastError || '未知错误');
        }
        return delivery.status;
    }

    /* --------------------------------------------------------------- runners */

    function providerStateText(provider) {
        if (!provider) return '';
        if (provider.connectionState === 'connected') {
            return '已验证 · ' + relativeTime(provider.lastTestedAt);
        }
        if (provider.connectionState === 'failed') {
            return '连接失败 · ' + ((provider.lastTestResult || {}).errorCode || '未知错误');
        }
        // §19.5: a dispatched smoke is still running; saying "待验证" here would
        // read as idle when an Action is actually in flight.
        if (provider.connectionState === 'testing') {
            return '冒烟运行中 · 等待 Action 回调';
        }
        return provider.lastTestedAt
            ? '待验证 · 上次测试 ' + relativeTime(provider.lastTestedAt)
            : '待验证';
    }

    function setConnectionBadge(node, connectionState, fallback) {
        if (fallback) {
            node.className = 'badge';
            node.textContent = fallback;
            return;
        }
        if (connectionState === 'connected') {
            node.className = 'badge green';
            node.textContent = '已连接';
        } else if (connectionState === 'failed') {
            node.className = 'badge red';
            node.textContent = '连接失败';
        } else if (connectionState === 'testing') {
            node.className = 'badge orange';
            node.textContent = '测试中';
        } else {
            node.className = 'badge';
            node.textContent = '待验证';
        }
    }

    function smokeHistoryStatus(entry) {
        if (entry.status === 'running') return '测试中';
        if (entry.ok === true) return '成功';
        return '失败 · ' + (entry.errorCode || '未知错误');
    }

    /* §19.5: block a malformed endpoint locally so the focus returns to the
       field before any test request is made. The server enforces it too. */
    function validateResponsesEndpoint(value) {
        var raw = String(value || '').trim();
        if (!raw) return false;
        try {
            var endpoint = new URL(raw);
            if (['http:', 'https:'].indexOf(endpoint.protocol) === -1) return false;
            var path = endpoint.pathname.replace(/\/+$/, '');
            return path.slice(-13) === '/v1/responses';
        } catch (error) {
            return false;
        }
    }

    document.addEventListener('click', function (event) {
        var viewButton = event.target.closest('[data-view]');
        if (viewButton) {
            switchView(viewButton.dataset.view);
            return;
        }

        var issueButton = event.target.closest('[data-issue]');
        if (issueButton) {
            selectIssue(issueButton.dataset.issue);
            $('queuePanel').classList.remove('open');
            $('mobileQueueButton').setAttribute('aria-expanded', 'false');
            return;
        }

        var mentionButton = event.target.closest('[data-mention]');
        if (mentionButton) {
            var input = $('replyInput');
            var mention = mentionButton.dataset.mention + ' ';
            if (input.value.indexOf(mentionButton.dataset.mention) === -1) {
                input.value = mention + input.value;
            }
            if ($('replyMode').querySelector('option[value="resume"]')) {
                $('replyMode').value = 'resume';
            }
            state.replyRequestId = '';
            input.focus();
            updateRouteHint();
            return;
        }

        var humanActionButton = event.target.closest('[data-human-action]');
        if (humanActionButton) {
            respondToHumanAction(humanActionButton);
            return;
        }

        var actionButton = event.target.closest('[data-action]');
        if (actionButton) {
            var action = actionButton.dataset.action;
            if (action === 'focus-reply') {
                $('replyInput').focus();
            } else if (action === 'close') {
                $('replyMode').value = 'close';
                updateRouteHint();
                $('replyInput').focus();
                toastTimer('请填写关闭理由后提交回复');
            } else if (action === 'reopen') {
                reopenIssue(actionButton);
            } else if (action === 'copy-owner-link') {
                copyOwnerReturnLink();
            } else if (action === 'save-classification') {
                saveClassification(actionButton);
            }
            return;
        }

        var defaultProviderButton = event.target.closest('[data-default-provider]');
        if (defaultProviderButton) {
            $('defaultRunnerSelect').value = defaultProviderButton.dataset.defaultProvider;
            state.runners.defaultProvider = defaultProviderButton.dataset.defaultProvider;
            renderRunners();
            markDirty('runners');
            return;
        }

        var replayButton = event.target.closest('[data-replay-delivery]');
        if (replayButton) {
            replayButton.disabled = true;
            replayButton.textContent = '重放中…';
            request(
                '/api/feedback/deliveries/' +
                    encodeURIComponent(replayButton.dataset.replayDelivery) +
                    '/replay',
                { method: 'POST', body: {} }
            )
                .then(function (payload) {
                    toastTimer(
                        payload.result.ok
                            ? '重放成功：' + payload.result.responseStatus
                            : '重放未成功：' + (payload.result.errorCode || '未知错误')
                    );
                    return loadAutomation();
                })
                .catch(function (error) {
                    replayButton.disabled = false;
                    replayButton.textContent = '重放';
                    handleAuthError(error);
                });
            return;
        }

        var testProviderButton = event.target.closest('[data-test-provider]');
        if (testProviderButton) {
            testProvider(testProviderButton);
            return;
        }

        var switchButton = event.target.closest('.switch');
        if (switchButton) {
            setSwitch(switchButton, !switchButton.classList.contains('on'));
            markDirty(switchButton.dataset.automationSwitch ? 'automation' : 'runners');
        }
    });

                if (!response.ok) throw new Error('密码不正确');
                return response.json();
            })
            .then(function (payload) {
                state.adminToken = payload.token;
                state.actor = 'admin';
                try {
                    window.sessionStorage.setItem(ADMIN_TOKEN_KEY, payload.token);
                } catch (error) {
                    /* ignore */
                }
                $('adminPassword').value = '';
                status.className = 'connection-status success';
                status.innerHTML = ICON_CHECK + '登录成功';
                applyActorChrome();
                switchView('issues');
                return refreshAll();
            })
            .catch(function (error) {
                status.className = 'connection-status error';
                status.textContent = error.message || '登录失败';
            })
            .finally(function () {
                submit.disabled = false;
                submit.textContent = '登录';
            });

    $('replyAttachments').addEventListener('change', function (event) {
        var files = Array.prototype.slice.call(event.target.files || []);
        if (files.length > 5) {
            toastTimer('最多选择 5 个附件');
            event.target.value = '';
            return;
        }
        var oversized = files.find(function (file) {
            return file.size > 4 * 1024 * 1024;
        });
        if (oversized) {
            toastTimer('单个附件不能超过 4 MiB');
            event.target.value = '';
            return;
        }
        if (estimatedReplyAttachmentBytes(files) > 18 * 1024 * 1024) {
            toastTimer('回复和附件的总请求不能超过 18 MiB');
            event.target.value = '';
            return;
        }
        Promise.all(files.map(readReplyAttachment))
            .then(function (attachments) {
                state.replyAttachments = attachments;
                renderReplyAttachments();
            })
            .catch(function () {
                toastTimer('附件读取失败，请重试');
                state.replyAttachments = [];
                event.target.value = '';
                renderReplyAttachments();
            });
    });

    $('replyForm').addEventListener('submit', function (event) {
        event.preventDefault();
        if (state.submitting || !state.issue) return;

        var input = $('replyInput');
        var body = input.value.trim();
        if (!body && !state.replyAttachments.length) {
            input.focus();
            toastTimer('请先填写回复内容');
            return;
        }

        var mode = $('replyMode').value;
        var requestId = state.replyRequestId || createCommentRequestId();
        var commentPayload = {
            body: body,
            mode: mode,
            expectedVersion: state.issue.version,
            requestId: requestId,
            attachments: state.replyAttachments,
        };
        if (new Blob([JSON.stringify(commentPayload)]).size > 18 * 1024 * 1024) {
            toastTimer('回复和附件的总请求不能超过 18 MiB');
            return;
        }
        state.replyRequestId = requestId;
        var submit = $('replySubmit');
        state.submitting = true;
        submit.disabled = true;
        submit.textContent = '提交中…';

        request('/api/feedback/issues/' + encodeURIComponent(state.issue.key) + '/comments', {
            method: 'POST',
            body: commentPayload,
        })
            .then(function (payload) {
                state.replyRequestId = '';
                input.value = '';
                state.replyAttachments = [];
                $('replyAttachments').value = '';
                renderReplyAttachments();
                setHidden($('replyPreview'), true);
                input.style.display = '';
                var success = $('replySuccess');
                success.style.display = 'inline-flex';
                success.innerHTML = ICON_CHECK + '已写入时间线';
                if (payload.mode !== payload.requestedMode) {
                    toastTimer('回复已记录；当前状态不支持唤醒处理者，本次仅写入时间线');
                } else if (payload.mode === 'resume') {
                    toastTimer(
                        '回复已发布，并唤醒 ' + (PROVIDER_LABELS[payload.provider] || '处理者')
                    );
                } else if (payload.mode === 'close') {
                    toastTimer('回复已发布，Issue 已关闭');
                } else {
                    toastTimer('回复已发布，本次未触发处理者');
                }
                return Promise.all([
                    selectIssue(state.issue.key),
                    state.actor === 'admin' ? loadIssues() : Promise.resolve(),
                ]);
            })
            .catch(function (error) {
                if (error && error.status === 409) {
                    state.replyRequestId = '';
                    toastTimer('该 Issue 已被更新，正在重新载入');
                    return selectIssue(state.issue.key);
                }
                handleAuthError(error);
                return null;
            })
            .finally(function () {
                state.submitting = false;
                submit.disabled = false;
                submit.textContent = '提交回复';
            });
    });

    function respondToHumanAction(button) {
        if (state.submitting) return;

        var decision = button.dataset.decision;
        var payload = { decision: decision };
        if (decision === 'ready_for_deploy') {
            if (!button.dataset.candidate) {
                toastTimer('缺少候选实现标识，无法批准交付');
                return;
            }
            payload.candidateId = button.dataset.candidate;
        }

        if (button.dataset.designDecision) {
            if (!button.dataset.design) {
                toastTimer('缺少方案版本标识，无法提交决定');
                return;
            }
            var noteInput = $('designDecisionNote');
            var note = noteInput ? noteInput.value.trim() : '';
            if (button.dataset.designDecision !== 'approve' && !note) {
                toastTimer('要求修订或拒绝时请填写具体原因');
                if (noteInput) noteInput.focus();
                return;
            }
            payload.designDecision = button.dataset.designDecision;
            payload.designId = button.dataset.design;
            payload.note = note;
        }

        // SCN-FWB-037：采纳这份分析并授权实施。备注可选——不写的话服务端会补一句
        // 默认说明，时间线上不会只剩一个状态跳变。
        if (button.dataset.policyDecision) {
            payload.policyDecision = button.dataset.policyDecision;
            var approvalInput = $('humanActionNote');
            var approvalNote = approvalInput ? approvalInput.value.trim() : '';
            if (approvalNote) payload.note = approvalNote;
        }

        // SCN-FWB-036: 没有新信息就重新处理，只会拿回逐字相同的结论。服务端会拒，
        // 这里先拦一道并把光标放到该填的地方——让用户在按钮上撞 409 是很差的解释方式。
        if (button.dataset.needsNote) {
            var followUpInput = $('humanActionNote');
            var followUp = followUpInput ? followUpInput.value.trim() : '';
            if (!followUp) {
                toastTimer('请先补充信息：没有新信息，重新处理会得到与上一轮相同的结论');
                if (followUpInput) followUpInput.focus();
                return;
            }
            payload.note = followUp;
        }
        state.submitting = true;
        setButtonBusy(button, '提交中…');
        // §19.1/SCN-FWB-025：提交期间用户仍然可以点开别的 Issue。快照按提交那一刻的
        // 选择代次记账，迟到的响应就不会把当前选中的详情覆盖回去。
        var decisionGeneration = state.selectionGeneration;
        request(
            '/api/feedback/human-actions/' +
                encodeURIComponent(button.dataset.humanAction) +
                '/respond',
            {
                method: 'POST',
                body: payload,
            }
        )
            .then(function (result) {
                if (result && result.resumeState === 'pending') {
                    toastTimer('方案决定已记录；Workflow 恢复待重试');
                } else {
                    toastTimer(
                        button.dataset.designDecision
                            ? '方案决定已提交'
                            : button.dataset.policyDecision === 'implement'
                              ? '已采纳这份分析，正在按结论开始实施'
                              : '已提交决定：' +
                                humanActionLabel(
                                    decision,
                                    state.humanAction && state.humanAction.type
                                )
                    );
                }
                // 决定改变的是同一个详情面板，服务端已经把改完之后的完整快照放在
                // 响应里了——再发一次 snapshot 只是把同一份数据重取一遍，却要用户
                // 多等一整个跨源往返。队列顺序也变了，但那不挡着看结果，放到后台刷。
                if (result && result.snapshot) {
                    applyIssueSnapshot(result.snapshot, decisionGeneration);
                    loadIssues();
                    return null;
                }
                return Promise.all([selectIssue(state.issue.key), loadIssues()]);
            })
            .catch(handleAuthError)
            .finally(function () {
                state.submitting = false;
                setButtonBusy(button, '');
            });
    }

            });
    }
